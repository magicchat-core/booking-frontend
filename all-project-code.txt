import { createContext, useContext, useMemo, useState } from 'react';
import { currentUser, saveSession, clearSession } from './client';
import api from './client';

const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(currentUser());

  async function login(email, password) {
    const { data } = await api.post('/auth/login', { email, password });
    saveSession(data);
    setUser({ user_id: data.user_id, email: data.email, role: data.role });
    return data;
  }

  async function register(email, password, role = 'staff') {
    const { data } = await api.post('/auth/register', { email, password, role });
    saveSession(data);
    setUser({ user_id: data.user_id, email: data.email, role: data.role });
    return data;
  }

  function logout() {
    clearSession();
    setUser(null);
  }

  const value = useMemo(() => ({ user, login, register, logout }), [user]);
  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

export function useAuth() {
  return useContext(AuthContext);
}

import axios from 'axios';

export const TOKEN_KEY = 'crm_token';
export const USER_KEY  = 'crm_user';

// Priority:
//   1. Runtime config injected via public/config.js  (works without rebuild)
//   2. Build-time env var VITE_API_BASE_full_URL          (baked in at build)
//   3. '/' (same origin — relies on dev proxy / prod reverse proxy)
const BASE_URL =
  window.__APP_CONFIG__?.API_BASE_full_URL ||
  import.meta.env.VITE_API_BASE_full_URL ||
  '/';

const api = axios.create({
  baseURL: BASE_URL,
  headers: { 'Content-Type': 'application/json' },
});

api.interceptors.request.use((config) => {
  const token = localStorage.getItem(TOKEN_KEY);
  if (token) config.headers.Authorization = `Bearer ${token}`;
  return config;
});

api.interceptors.response.use(
  (r) => r,
  (err) => {
    if (err.response?.status === 401) {
      localStorage.removeItem(TOKEN_KEY);
      localStorage.removeItem(USER_KEY);
      if (!window.location.pathname.startsWith('/login')) {
        window.location.href = '/login';
      }
    }
    return Promise.reject(err);
  }
);

export function saveSession({ token, user_id, email, role }) {
  localStorage.setItem(TOKEN_KEY, token);
  localStorage.setItem(USER_KEY, JSON.stringify({ user_id, email, role }));
}

export function clearSession() {
  localStorage.removeItem(TOKEN_KEY);
  localStorage.removeItem(USER_KEY);
}

export function currentUser() {
  try {
    return JSON.parse(localStorage.getItem(USER_KEY) || 'null');
  } catch { return null; }
}

export default api;

// ============================================================
// Sageion — reference client integration.
//
// The SDK exposes two methods on window.sageion_os:
//
//   setUp()       — configuration. Call ONCE per page load.
//   initialize()  — auth state. Call whenever the user changes.
//
// Four rules a correct integration must honor:
//
//   1. setUp() runs once, awaited, before any initialize().
//   2. initialize() is called from exactly one place.
//   3. initialize() is gated on auth having SETTLED
//      (loading === false), not on user being non-null.
//   4. initialize() receives the current user, once per change.
//
// Two effects. Two concerns. No overlap. No state refs.
// ============================================================

function useSageionSetUp() {
  const [ready, setReady] = useState(false);
  const ran = useRef(false);

  useEffect(() => {
    if (ran.current) return;
    ran.current = true;

    (async () => {
      try {
        await window.sageion_os.setUp(
          window.__APP_CONFIG__.SAGEION_APP_ID,
          window.__APP_CONFIG__.SAGEION_API_KEY,
          window.__APP_CONFIG__.SAGEION_REGION,
          window.__APP_CONFIG__.ROOT_SAGEION_ID,
        );
        setReady(true);
      } catch (err) {
        // The SDK renders its own user-facing error UI for
        // credential or network failures. Log and move on.
        console.error('[Sageion] setUp failed:', err);
      }
    })();
  }, []);

  return ready;
}

function useSageionInitialize(setUpReady) {
  const { user, loading } = useAuth();

  useEffect(() => {
    // Gate 1: setUp() must have resolved.
    if (!setUpReady) return;

    // Gate 2: AuthContext must have settled. Without this, the
    // effect fires while `user` is still null during the initial
    // auth check, producing an anonymous → authenticated
    // transition on every page load.
    if (loading) return;

    // The user's current identity. The SDK handles every
    // transition internally:
    //
    //   user set   → initialize({ uid })  → login / switch
    //   user null  → initialize({})       → logout → anonymous
    //
    // Same payload twice → no-op. Safe to call as often as this
    // effect fires.
    //
    // NOTE: the CRM's user object carries `user_id` (not `id`).
    window.sageion_os
      .initialize(user ? { uid: String(user.user_id) } : {})
      .catch((err) => console.error('[Sageion] initialize failed:', err));
  }, [setUpReady, loading, user]);
}

// Must be rendered INSIDE <AuthProvider> so useAuth() resolves
// to the correct context.
function SageionBridge() {
  const setUpReady = useSageionSetUp();
  useSageionInitialize(setUpReady);
  return null;
}

function Shell({ children }) {
  return <ProtectedRoute><Layout>{children}</Layout></ProtectedRoute>;
}

import { useState } from 'react';
import api from '../api/client';

export default function Customers() {
  const [phone, setPhone] = useState('+911111111111');
  const [customer, setCustomer] = useState(null);
  const [error, setError] = useState('');
  const [newCust, setNewCust] = useState({ name: '', phone: '' });
  const [createMode, setCreateMode] = useState(false);
  const [delta, setDelta] = useState(100);
  const [reason, setReason] = useState('birthday');
  const [msg, setMsg] = useState('');

  async function search(e) {
    if (e) e.preventDefault();
    setError(''); setMsg(''); setCustomer(null); setCreateMode(false);
    try {
      const { data } = await api.get(`/customers?phone=${encodeURIComponent(phone)}`);
      setCustomer(data);
    } catch (err) {
      const status = err.response?.status;
      if (status === 404) {
        setCreateMode(true);
        setNewCust({ name: '', phone });
        setMsg('No customer with this phone — fill in a name to register.');
      } else {
        setError(err.response?.data?.error || err.message);
      }
    }
  }

  async function create(e) {
    e.preventDefault();
    setError(''); setMsg('');
    try {
      const { data } = await api.post('/customers', newCust);
      setCustomer(data);
      setCreateMode(false);
      setMsg(`Created ${data.customer_id}.`);
    } catch (err) {
      setError(err.response?.data?.error || err.message);
    }
  }

  async function adjustPoints(e) {
    e.preventDefault();
    setError(''); setMsg('');
    try {
      const { data } = await api.patch(`/customers/${customer.customer_id}/loyalty-points`, {
        delta: Number(delta),
        reason,
      });
      setCustomer({ ...customer, loyalty_points: data.new_total });
      setMsg(`Points updated to ${data.new_total}.`);
    } catch (err) {
      setError(err.response?.data?.error || err.message);
    }
  }

  return (
    <div>
      <h1>Customers</h1>
      <p className="muted">Search by phone. If not found, register, then add loyalty points (Workflow 5).</p>

      <form className="inline-form" onSubmit={search}>
        <input value={phone} onChange={(e) => setPhone(e.target.value)} placeholder="+911111111111" />
        <button className="btn btn-primary">Search</button>
      </form>

      {error && <div className="alert error">{error}</div>}
      {msg && <div className="alert info">{msg}</div>}

      {createMode && (
        <form className="card" onSubmit={create}>
          <h3>Register new customer</h3>
          <label>Name
            <input value={newCust.name} onChange={(e) => setNewCust({ ...newCust, name: e.target.value })} required />
          </label>
          <label>Phone
            <input value={newCust.phone} onChange={(e) => setNewCust({ ...newCust, phone: e.target.value })} required />
          </label>
          <button className="btn btn-primary">Create</button>
        </form>
      )}

      {customer && (
        <div className="card">
          <h2>{customer.name}</h2>
          <p className="mono muted">{customer.customer_id} · {customer.phone}</p>
          <p className="stat-num">{customer.loyalty_points} <span className="stat-label">loyalty points</span></p>
          <form className="inline-form" onSubmit={adjustPoints}>
            <label>Delta
              <input type="number" value={delta} onChange={(e) => setDelta(e.target.value)} />
            </label>
            <label>Reason
              <input value={reason} onChange={(e) => setReason(e.target.value)} />
            </label>
            <button className="btn btn-primary">Adjust points</button>
          </form>
        </div>
      )}
    </div>
  );
}

import { useEffect, useState } from 'react';
import api from '../api/client';

export default function Inventory() {
  const [items, setItems] = useState([]);
  const [error, setError] = useState('');

  async function load() {
    try {
      const { data } = await api.get('/inventory');
      setItems(data);
    } catch (err) { setError(err.response?.data?.error || err.message); }
  }
  useEffect(() => { load(); }, []);

  async function adjust(item) {
    const raw = prompt(`Adjust ${item.name} by (e.g. -2 or 5):`, '1');
    if (raw === null) return;
    const n = Number(raw);
    if (!Number.isFinite(n)) return alert('Not a number');
    try {
      const { data } = await api.patch(`/inventory/${item.material_id}`, {
        adjust_qty: n, reason: 'manual adjust from UI',
      });
      setItems((prev) => prev.map((i) => i.material_id === item.material_id
        ? { ...i, current_qty: data.new_qty } : i));
    } catch (err) { alert(err.response?.data?.error || err.message); }
  }

  return (
    <div>
      <h1>Inventory</h1>
      <p className="muted">Materials below threshold are highlighted. Adjust quantity to correct stock manually.</p>
      {error && <div className="alert error">{error}</div>}
      <table className="table">
        <thead>
          <tr><th>Material</th><th>Qty</th><th>Unit</th><th>Low threshold</th><th></th></tr>
        </thead>
        <tbody>
          {items.map((i) => {
            const low = i.current_qty < i.low_threshold;
            return (
              <tr key={i.material_id} className={low ? 'row-warn' : ''}>
                <td><strong>{i.name}</strong><br/><span className="mono muted">{i.material_id}</span></td>
                <td className={low ? 'warn' : ''}>{i.current_qty}</td>
                <td>{i.unit}</td>
                <td>{i.low_threshold}</td>
                <td><button className="btn btn-small" onClick={() => adjust(i)}>Adjust</button></td>
              </tr>
            );
          })}
        </tbody>
      </table>
    </div>
  );
}

export default function Orders() {

  const [error, setError] = useState('');
  const [note, setNote] = useState('');

    if (e) e.preventDefault();
    setError(''); setOrder(null); setNoteMsg('');
    try {
      const { data } = await api.get(`/orders/${orderId}`);
      setOrder(data);
    } catch (err) {
      setError(err.response?.data?.error || err.message);
    }
  }

  async function cancel() {
    setError(''); setNoteMsg('');
    try {
      const { data } = await api.post(`/orders/${orderId}/cancel`);
      setOrder({ ...order, status: data.status });
    } catch (err) {
      const status = err.response?.status;
      const msg = err.response?.data?.error || err.message;
      setError(msg);
      if (status === 409) {
        // Branch: log a support note
        try {
          await api.post('/support-notes', {
            order_id: orderId,
            note: `Cancel rejected (${msg}). Customer informed; escalation logged.`,
          });
          setNoteMsg('Cancellation was rejected — a support note was logged automatically.');
        } catch (noteErr) {
          setNoteMsg('Could not log support note: ' + noteErr.message);
        }
      }
    }
  }

  async function addNote(e) {
    e.preventDefault();
    setNoteMsg('');
    try {
      await api.post('/support-notes', { order_id: orderId, note });
      setNoteMsg('Support note added.');
      setNote('');
    } catch (err) {

import { useEffect, useState } from 'react';
import api from '../api/client';

export default function PurchaseOrders() {
  const [pos, setPOs] = useState([]);
  const [materials, setMaterials] = useState([]);
  const [form, setForm] = useState({ supplier: 'FreshMeat', material_id: 'chicken', qty: 10, unit: 'kg' });
  const [error, setError] = useState('');
  const [busy, setBusy] = useState(false);

  async function load() {
    try {
      const [p, inv] = await Promise.all([api.get('/purchase-orders'), api.get('/inventory')]);
      setPOs(p.data);
      setMaterials(inv.data);
      if (inv.data.length && !inv.data.some((m) => m.material_id === form.material_id)) {
        setForm((f) => ({ ...f, material_id: inv.data[0].material_id, unit: inv.data[0].unit }));
      }
    } catch (err) { setError(err.response?.data?.error || err.message); }
  }
  useEffect(() => { load(); }, []); // eslint-disable-line react-hooks/exhaustive-deps

  async function create(e) {
    e.preventDefault();
    setBusy(true); setError('');
    try {
      await api.post('/purchase-orders', { ...form, qty: Number(form.qty) });
      await load();
    } catch (err) { setError(err.response?.data?.error || err.message); }
    finally { setBusy(false); }
  }

  async function approve(po, approved) {
    try {
      const { data } = await api.patch(`/purchase-orders/${po.po_id}/approve`, { approved });
      await load();
      if (approved) {
        alert(`Approved. Inventory updated: ${data.inventory_updated ? 'yes' : 'no (already processed)'}`);
      }
    } catch (err) { alert(err.response?.data?.error || err.message); }
  }

  function pickMaterial(id) {
    const m = materials.find((x) => x.material_id === id);
    setForm((f) => ({ ...f, material_id: id, unit: m?.unit || f.unit }));
  }

  return (
    <div>
      <h1>Purchase Orders</h1>
      <p className="muted">Create a PO, then approve it — inventory is credited atomically (Workflows 2 &amp; 3).</p>
      {error && <div className="alert error">{error}</div>}

      <form className="card inline-form" onSubmit={create}>
        <label>Supplier
          <input value={form.supplier} onChange={(e) => setForm({ ...form, supplier: e.target.value })} />
        </label>
        <label>Material
          <select value={form.material_id} onChange={(e) => pickMaterial(e.target.value)}>
            {materials.map((m) => (
              <option key={m.material_id} value={m.material_id}>{m.name} ({m.material_id})</option>
            ))}
          </select>
        </label>
        <label>Qty
          <input type="number" step="0.01" value={form.qty} onChange={(e) => setForm({ ...form, qty: e.target.value })} />
        </label>
        <label>Unit
          <input value={form.unit} onChange={(e) => setForm({ ...form, unit: e.target.value })} />
        </label>
        <button className="btn btn-primary" disabled={busy}>{busy ? '…' : 'Create PO'}</button>
      </form>

      <table className="table">
        <thead>
          <tr><th>PO</th><th>Supplier</th><th>Material</th><th>Qty</th><th>Status</th><th>Actions</th></tr>
        </thead>
        <tbody>
          {pos.map((p) => (
            <tr key={p.po_id}>
              <td className="mono">{p.po_id}</td>
              <td>{p.supplier}</td>
              <td>{p.material_id}</td>
              <td>{p.qty} {p.unit}</td>
              <td><span className={`pill ${p.status}`}>{p.status}</span></td>
              <td>
                {p.status === 'pending_supplier' ? (
                  <>
                    <button className="btn btn-small btn-primary" onClick={() => approve(p, true)}>Approve</button>{' '}
                    <button className="btn btn-small btn-danger" onClick={() => approve(p, false)}>Reject</button>
                  </>
                ) : <span className="muted">—</span>}
              </td>
            </tr>
          ))}
          {pos.length === 0 && (
            <tr><td colSpan="6" className="muted">No purchase orders yet.</td></tr>
          )}
        </tbody>
      </table>
    </div>
  );
}

import { useEffect, useState, useRef } from 'react';
import api from '../api/client';

export default function RunningTables() {
  const [tables, setTables] = useState([]);
  const [activeId, setActiveId] = useState(null);
  const [messages, setMessages] = useState([]);   // chat transcript for the active table
  const [input, setInput] = useState('');
  const [busy, setBusy] = useState(false);
  const [err, setErr] = useState('');
  const endRef = useRef(null);

  async function loadTables() {
    try {
      const { data } = await api.get('/tables');
      setTables(data);
      if (data.length && !activeId) setActiveId(data[0].table_id);
    } catch (e) {
      setErr(e.response?.data?.error || e.message);
    }
  }
  useEffect(() => { loadTables(); }, []); // eslint-disable-line react-hooks/exhaustive-deps
  useEffect(() => { endRef.current?.scrollIntoView({ behavior: 'smooth' }); }, [messages]);

  const active = tables.find((t) => t.table_id === activeId) || null;

  async function handleSend(e) {
    e.preventDefault();
    if (!input.trim() || !active) return;
    const text = input.trim();
    setInput('');
    setMessages((m) => [...m, { role: 'user', text }]);
    setBusy(true); setErr('');

    try {
      // Very small intent matcher — the demo uses two verbs:
      //   "apply the <tag> discount to <table>"  → POST /tables/:id/discount
      //   anything else                          → POST /tables/:id/ask
      const lower = text.toLowerCase();
      const applyMatch = lower.match(/apply\s+(?:the\s+)?(\w+)\s+discount/);

      if (applyMatch) {
        const tag = applyMatch[1];
        const { data } = await api.post(`/tables/${active.table_id}/discount`, { tag });
        setMessages((m) => [
          ...m,
          {
            role: 'assistant',
            text:
              `Applied ${data.tag} ${data.pct}% discount.\n` +
              `Discountable items:\n` +
              data.breakdown.map((b) => `  • ${b.name} ₹${b.line_total.toFixed(2)} → ₹${b.discount.toFixed(2)} off`).join('\n') +
              (data.excluded.length
                ? `\nExcluded:\n` + data.excluded.map((x) => `  • ${x.name} ₹${x.line_total.toFixed(2)} (${x.reason})`).join('\n')
                : '') +
              `\nNew running amount: ₹${data.new_running_total.toFixed(2)}`,
          },
        ]);
        await loadTables();
      } else {
        const { data } = await api.post(`/tables/${active.table_id}/ask`, { question: text });
        setMessages((m) => [...m, { role: 'assistant', text: data.answer }]);
      }
    } catch (e) {
      const msg = e.response?.data?.error || e.message;
      setMessages((m) => [...m, { role: 'assistant', text: `⚠ ${msg}` }]);
    } finally {
      setBusy(false);
    }
  }

  return (
    <div>
      <h1>Running Tables</h1>
      <p className="muted">Owner view. Open tables are listed on the left; each one has a chat box. Try: <code>apply the Friends discount to T6</code>.</p>
      {err && <div className="alert error">{err}</div>}

      <div className="tables-grid">
        {/* Left column: table list */}
        <div className="card tables-list">
          <h3>Open tables</h3>
          {tables.length === 0 && <p className="muted">No open tables.</p>}
          {tables.map((t) => (
            <button
              key={t.table_id}
              className={`table-row ${t.table_id === activeId ? 'active' : ''}`}
              onClick={() => { setActiveId(t.table_id); setMessages([]); }}
            >
              <div className="row-between">
                <strong>{t.table_id}</strong>
                <span className="mono">₹{t.running_total.toFixed(2)}</span>
              </div>
              <div className="muted small">
                {t.items.length} item{t.items.length === 1 ? '' : 's'}
                {t.customer_id ? ` · ${t.customer_id}` : ' · walk-in'}
              </div>
            </button>
          ))}
        </div>

        {/* Right column: active table detail + chat */}
        <div className="card">
          {active ? (
            <>
              <h2>{active.table_id} <span className="muted mono small">{active.customer_id || 'walk-in'}</span></h2>

              <table className="table">
                <thead>
                  <tr><th>Item</th><th>Qty</th><th>Rate</th><th>Line</th><th></th></tr>
                </thead>
                <tbody>
                  {active.items.map((it) => (
                    <tr key={it.line_id}>
                      <td>{it.name}</td>
                      <td>{it.qty}</td>
                      <td>₹{it.unit_price.toFixed(2)}</td>
                      <td>₹{it.line_total.toFixed(2)}</td>
                      <td>
                        {it.ignore_discount && <span className="pill off">no discount</span>}
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>

              <div className="row-between totals">
                <span>Subtotal</span>
                <span className="mono">₹{active.subtotal.toFixed(2)}</span>
              </div>
              {active.last_discount && (
                <div className="row-between totals">
                  <span>{active.last_discount.tag} ({active.last_discount.pct}%)</span>
                  <span className="mono">−₹{Number(active.last_discount.amount).toFixed(2)}</span>
                </div>
              )}
              <div className="row-between totals grand">
                <span>Running total</span>
                <span className="mono">₹{active.running_total.toFixed(2)}</span>
              </div>

              <hr />
              <h3>Chat</h3>
              <div className="chat-log">
                {messages.length === 0 && (
                  <p className="muted small">
                    Try: <code>apply the Friends discount to {active.table_id}</code>, or <code>what was excluded?</code>
                  </p>
                )}
                {messages.map((m, i) => (
                  <div key={i} className={`chat-msg ${m.role}`}>
                    <div className="chat-role">{m.role}</div>
                    <pre className="chat-text">{m.text}</pre>
                  </div>
                ))}
                <div ref={endRef} />
              </div>
              <form className="inline-form" onSubmit={handleSend}>
                <input
                  style={{ flex: 1 }}
                  value={input}
                  onChange={(e) => setInput(e.target.value)}
                  placeholder={`Type a command…`}
                  disabled={busy}
                />
                <button className="btn btn-primary" disabled={busy || !input.trim()}>
                  {busy ? '…' : 'Send'}
                </button>
              </form>
            </>
          ) : (
            <p className="muted">Select a table to see its order and chat.</p>
          )}
        </div>
      </div>
    </div>
  );
}
